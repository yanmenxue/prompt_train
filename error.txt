import urllib.request, json, ssl                                                                                                                               
ctx = ssl.create_default_context(); ctx.check_hostname=False; ctx.verify_mode=ssl.CERT_NONE                                                                    
url='https://api.github.com/repos/huggingface/transformers/contents/src/transformers/models/qwen3?ref=main'                                                    
try:                                                                                                                                                           
  d=json.load(urllib.request.urlopen(url, context=ctx))                                                                                                      
  print('files:', [f['name'] for f in d])                                                                                                                    
except Exception as e:                                                                                                                                         
  print('err:', e)
