import sys
import json
import urllib.request
import ssl  # 导入 ssl 模块

# 1. 创建一个不验证证书的 SSL 上下文
ssl_context = ssl.create_default_context()
ssl_context.check_hostname = False
ssl_context.verify_mode = ssl.CERT_NONE

url = 'https://api.github.com/repos/huggingface/transformers/commits?path=src/transformers/models/qwen3/modeling_qwen3.py&per_page=3'

# 2. 在 urlopen 中传入 context 参数
with urllib.request.urlopen(url, context=ssl_context) as response:
    d = json.load(response)
    for c in d:
        print(c["sha"][:8], c["commit"]["author"]["date"], c["commit"]["message"].split("\n")[0])
